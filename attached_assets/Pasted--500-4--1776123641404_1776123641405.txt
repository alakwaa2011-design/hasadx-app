// الحصول على جميع الفئات
function getCategories() {
return […new Set(questions.map(q => q.category))];
}

// الحصول على أسئلة حسب الفئة
function getByCategory(category) {
return questions.filter(q => q.category === category);
}

// الحصول على أسئلة حسب الصعوبة
function getByDifficulty(difficulty) {
return questions.filter(q => q.difficulty === difficulty);
}

// اختيار سؤال عشوائي
function getRandomQuestion() {
return questions[Math.floor(Math.random() * questions.length)];
}

// اختيار n سؤال عشوائي بدون تكرار
function getRandomQuestions(n) {
const shuffled = […questions].sort(() => Math.random() - 0.5);
return shuffled.slice(0, n);
}

// إحصائيات البنك
function getStats() {
const cats = getCategories();
const stats = { total: questions.length, byCategory: {}, byDifficulty: {} };
cats.forEach(cat => stats.byCategory[cat] = getByCategory(cat).length);
[‘سهل’, ‘متوسط’, ‘صعب’].forEach(d => stats.byDifficulty[d] = getByDifficulty(d).length);
return stats;
}

module.exports = { questions, getCategories, getByCategory, getByDifficulty, getRandomQuestion, getRandomQuestions, getStats };
