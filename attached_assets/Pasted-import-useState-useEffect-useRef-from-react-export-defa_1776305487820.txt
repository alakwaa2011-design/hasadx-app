import { useState, useEffect, useRef } from "react";

export default function GameLobby() {
  const [navMode, setNavMode] = useState<"auto" | "manual">("auto");
  const [gifts, setGifts] = useState(false);
  const [hack, setHack] = useState(false);
  const [voice, setVoice] = useState(true);
  const [bots, setBots] = useState(4);
  const [codeCopied, setCodeCopied] = useState(false);
  const [linkCopied, setLinkCopied] = useState(false);
  const [hackText, setHackText] = useState("لعبة الاختراق");
  const [hackSub, setHackSub] = useState("صناديق وكلمات سرية");
  const glitchRef = useRef<ReturnType<typeof setInterval> | null>(null);

  const GLITCH = ['#', '@', '!', '%', '&', '*', '0', '1', 'X', '?'];

  function glitchAnimate(target: string) {
    let i = 0;
    if (glitchRef.current) clearInterval(glitchRef.current);
    glitchRef.current = setInterval(() => {
      const result = target.split("").map((c, idx) =>
        idx < i ? c : Math.random() > 0.5 ? GLITCH[Math.floor(Math.random() * GLITCH.length)] : c
      ).join("");
      setHackText(result);
      i++;
      if (i > target.length) {
        setHackText(target);
        clearInterval(glitchRef.current!);
      }
    }, 40);
  }

  function toggleHack() {
    if (!hack) {
      setHack(true);
      glitchAnimate("H4CK_M0DE_ON");
      setHackSub("> SYSTEM BREACH ACTIVATED_");
    } else {
      setHack(false);
      if (glitchRef.current) clearInterval(glitchRef.current);
      setHackText("لعبة الاختراق");
      setHackSub("صناديق وكلمات سرية");
    }
  }

  useEffect(() => () => { if (glitchRef.current) clearInterval(glitchRef.current); }, []);

  const s = makeStyles(hack);

  return (
    <div style={s.page}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Tajawal:wght@400;600;800&family=Share+Tech+Mono&display=swap');
        * { box-sizing: border-box; margin: 0; padding: 0; }
        @keyframes scan { 0%{top:0} 100%{top:100%} }
        input[type=range] { accent-color: #5B5EF4; }
      `}</style>

      {/* ─── Header ─── */}
      <div style={s.header}>
        <div style={s.logo}>
          <div style={s.logoDot}>H</div>
          <span>حصاد</span>
        </div>
        <span style={s.liveTag}>● مباشر</span>
      </div>

      <div style={s.body}>

        {/* ─── Code Box ─── */}
        <div style={s.codeBox}>
          <div style={s.codeRow}>
            <div style={{ flex: 1 }}>
              <div style={s.codeLabel}>كود الانضمام للعبة</div>
              <div style={s.codeNum}>716431</div>
              <div style={s.codeActions}>
                <button style={s.codeBtn}
                  onClick={() => { setCodeCopied(true); setTimeout(() => setCodeCopied(false), 1500); }}>
                  {codeCopied ? "✓ تم النسخ" : "نسخ الكود"}
                </button>
                <button style={s.codeBtn}
                  onClick={() => { setLinkCopied(true); setTimeout(() => setLinkCopied(false), 1500); }}>
                  {linkCopied ? "✓ تم!" : "🔗 نسخ الرابط"}
                </button>
              </div>
            </div>
            <div style={s.qrWrap}>
              <svg viewBox="0 0 56 56" width="58" height="58">
                <rect width="56" height="56" fill="white"/>
                <rect x="3" y="3" width="20" height="20" fill="none" stroke="#1A1F2E" strokeWidth="2"/>
                <rect x="7" y="7" width="12" height="12" fill="#1A1F2E"/>
                <rect x="33" y="3" width="20" height="20" fill="none" stroke="#1A1F2E" strokeWidth="2"/>
                <rect x="37" y="7" width="12" height="12" fill="#1A1F2E"/>
                <rect x="3" y="33" width="20" height="20" fill="none" stroke="#1A1F2E" strokeWidth="2"/>
                <rect x="7" y="37" width="12" height="12" fill="#1A1F2E"/>
                <rect x="33" y="33" width="5" height="5" fill="#1A1F2E"/>
                <rect x="40" y="33" width="5" height="5" fill="#1A1F2E"/>
                <rect x="48" y="33" width="5" height="5" fill="#1A1F2E"/>
                <rect x="33" y="40" width="5" height="5" fill="#1A1F2E"/>
                <rect x="40" y="48" width="5" height="5" fill="#1A1F2E"/>
                <rect x="48" y="40" width="5" height="5" fill="#1A1F2E"/>
                <rect x="48" y="48" width="5" height="5" fill="#1A1F2E"/>
              </svg>
            </div>
          </div>
        </div>

        {/* ─── Mode + Nav ─── */}
        <div style={s.row2}>
          <div style={s.miniCard}>
            <div style={s.mcLabel}>وضع اللعب</div>
            <div style={s.modeSel}>
              <span style={{ fontSize: 15 }}>👤</span>
              <div>
                <div style={s.modeName}>فردي</div>
                <div style={s.modeSub}>كل لاعب لوحده</div>
              </div>
            </div>
          </div>
          <div style={s.miniCard}>
            <div style={s.mcLabel}>التنقل</div>
            <div style={s.navBtns}>
              <button style={{ ...s.navBtn, ...(navMode === "auto" ? s.navBtnOn : {}) }} onClick={() => setNavMode("auto")}>تلقائي</button>
              <button style={{ ...s.navBtn, ...(navMode === "manual" ? s.navBtnOn : {}) }} onClick={() => setNavMode("manual")}>يدوي</button>
            </div>
          </div>
        </div>

        {/* ─── Features ─── */}
        <div style={s.feats}>

          <div style={s.feat}>
            <div style={s.featLeft}>
              <span style={s.featIc}>🎁</span>
              <div>
                <div style={s.featName}>الهدايا</div>
                <div style={s.featSub}>هدية كل 3 إجابات صحيحة</div>
              </div>
            </div>
            <Toggle on={gifts} onToggle={() => setGifts(!gifts)} />
          </div>

          {/* Hack row */}
          <div style={s.hackRow}>
            {hack && <div style={s.scanLine} />}
            <div style={s.featLeft}>
              <span style={s.featIc}>{hack ? "💻" : "🔐"}</span>
              <div>
                <div style={{ ...s.featName, ...(hack ? s.hackNameStyle : {}) }}>{hackText}</div>
                <div style={{ ...s.featSub,  ...(hack ? s.hackSubStyle  : {}) }}>{hackSub}</div>
              </div>
            </div>
            <Toggle on={hack} onToggle={toggleHack} hackMode={hack} />
          </div>

          <div style={s.feat}>
            <div style={s.featLeft}>
              <span style={s.featIc}>🔊</span>
              <div>
                <div style={s.featName}>قراءة صوتية</div>
                <div style={s.featSub}>تلاوة الأسئلة للطلاب</div>
              </div>
            </div>
            <Toggle on={voice} onToggle={() => setVoice(!voice)} />
          </div>

          <div style={{ ...s.feat, borderBottom: "none" }}>
            <div style={s.featLeft}>
              <span style={s.featIc}>🤖</span>
              <div>
                <div style={s.featName}>لاعبون وهميون</div>
                <div style={s.featSub}>{bots} لاعبين وهميين</div>
              </div>
            </div>
            <input type="range" min={1} max={10} value={bots} step={1}
              style={{ width: 70 }} onChange={e => setBots(Number(e.target.value))} />
          </div>

        </div>

        {/* ─── Players ─── */}
        <div style={s.playersRow}>
          <div>
            <div style={s.plTitle}>👥 اللاعبون المتصلون</div>
            <div style={s.plSub}>في انتظار الانضمام...</div>
          </div>
          <span style={{ fontSize: 22, fontWeight: 800, color: "#DDE1ED" }}>0</span>
        </div>

        {/* ─── Action Buttons ─── */}
        <div style={s.btnsRow}>
          <button style={s.btnCancel}>إلغاء</button>
          <button style={s.btnStart}>▶ ابدأ اللعبة!</button>
        </div>

      </div>
    </div>
  );
}

// ─── Toggle Component ───────────────────────
function Toggle({ on, onToggle, hackMode = false }: {
  on: boolean; onToggle: () => void; hackMode?: boolean;
}) {
  return (
    <div onClick={onToggle} style={{
      width: 36, height: 20, borderRadius: 10, cursor: "pointer",
      position: "relative", flexShrink: 0, transition: "background .25s",
      background: hackMode ? "#00FF41" : on ? "#5B5EF4" : "#DDE1ED",
    }}>
      <div style={{
        position: "absolute", top: 2, width: 16, height: 16,
        background: "#fff", borderRadius: "50%", transition: "right .25s",
        right: on ? 18 : 2, boxShadow: "0 1px 3px rgba(0,0,0,.2)",
      }} />
    </div>
  );
}

  return {
